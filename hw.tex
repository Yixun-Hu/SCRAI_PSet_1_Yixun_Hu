import inspect
from google.colab import files      # Colab only; drop outside Colab.

SUBMIT = ["steer"]                  # Functions to show in your write-up.

for name in SUBMIT:
    fn = globals().get(name)
    if fn is not None:
        with open(f"{name}.py", "w") as f:
            f.write(inspect.getsource(fn))
    files.download(f"{name}.py")    # Colab only; drop outside Colab.
